Introdução e criação de coordenadas

Neste projeto, vamos explorar a criação de gráficos e animações utilizando HTML e JavaScript. Através do elemento canvas 2D do HTML, podemos desenhar formas, imagens e até mesmo criar interações dinâmicas com o usuário. Este projeto serve como uma introdução à computação gráfica, permitindo que você compreenda os conceitos básicos e comece a desenvolver suas próprias aplicações visuais na web.

Um aspecto importante a ser considerado, antes de mais nada, é que precisaremos de uma "base comum" em HTML que poderá ser usada em todos os projetos. Essa base comum inclui a estrutura básica do HTML, a inclusão do elemento canvas e a configuração inicial do JavaScript. A seguir, apresentaremos um exemplo de código que serve como ponto de partida para todos os projetos subsequentes. 

<!DOCTYPE html>
<html lang="en">

    <meta charset="UTF-8">

    <title>Document</title>
</head>
    <body>
        <canvas id="myCanvas" style="border: 1px solid black;"></canvas>
        <script src="script.js"></script>
    </body>

Abaixo então temos o nosso primeiro código em JavaScript, que faz duas coisas importantes: desenha o plano cartesiano e desenha um quadrado de lado 1, com vértices nos pontos A(0, 0), B(1, 0), C(1, 1) e D(0, 1). O código é comentado para facilitar a compreensão. 
/** @type { HTMLCanvasElement } */
const c = document.getElementById("myCanvas")
const ctx = c.getContext("2d")

const width = 800
const height = 800
c.width = width
c.height = height
const step = 100
// Vamos considerar que 100 pixels equivale a 1 unidade matemática
const cx = width/2
const cy = height/2

function converterCoordenadas(xMatematico, yMatematico) {

    const xCanvas = cx + (xMatematico * step);
    const yCanvas = cy - (yMatematico * step);

    return { x: xCanvas, y: yCanvas };
}

function desenharQuadradoNovo() {
    ctx.beginPath()
    ctx.lineWidth=4
    ctx.strokeStyle = "green";
    // Pegamos as coordenadas processadas pela nossa função
    let p1 = converterCoordenadas(0, 0);
    let p2 = converterCoordenadas(1, 0);
    let p3 = converterCoordenadas(1, 1);
    let p4 = converterCoordenadas(0, 1);
    // Desenhamos usando os valores prontos, sem pensar em pixels!
    ctx.moveTo(p1.x, p1.y); // A(0, 0)
    ctx.lineTo(p2.x, p2.y); // B(1, 0)
    ctx.lineTo(p3.x, p3.y); // C(1, 1)
    ctx.lineTo(p4.x, p4.y); // D(0, 1)
    ctx.lineTo(p1.x, p1.y); // Volta para a origem: A(0, 0)
    ctx.closePath()
    ctx.stroke();
}

function planoCartesiano() {
    ctx.beginPath()
    ctx.lineWidth = 1
    ctx.strokeStyle="black"

    for (let x=0; x<=width; x+=step) {
        ctx.moveTo(x, 0);
        ctx.lineTo(x, height)
    }

    for (let y=0; y<=height; y+=step) {
        ctx.moveTo(0, y)
        ctx.lineTo(width, y)
    }
    ctx.stroke()
}

function desenharEixos() {
    ctx.beginPath()
    ctx.moveTo(width/2, 0)
    ctx.lineTo(width/2, height)
    ctx.lineWidth = 2
    ctx.strokeStyle="red"
    ctx.stroke()
    ctx.closePath()

    ctx.beginPath()
    ctx.moveTo(0, height/2)
    ctx.lineTo(width, height/2)
    ctx.lineWidth = 2
    ctx.strokeStyle="red"
    ctx.stroke()
    ctx.closePath()
}

Agora que já temos as coordenadas funcionando, vamos criar um círculo usando somente linhas. O código abaixo mostra como fazer isso, desenhando o círculo com um passo de 45 graus. 

/** @type {HTMLCanvasElement} */
const c = document.getElementById("myCanvas")
const ctx = c.getContext("2d")

c.width = 800
c.height = 800
const width = c.width
const height = c.height

// Agora vamos criar um círculo usando somente linhas
const centerX = width/2
const centerY = height/2
const radius = 200
ctx.beginPath()
for (let angle = 0; angle <= 360; angle=angle+45) {
    const x = centerX + radius * Math.cos(angle * Math.PI/180)
    const y = centerY + radius * Math.sin(angle * Math.PI/180)
    ctx.lineTo(x, y)
}
ctx.stroke()

const step = 100
function desenharPlanoCartesiano() {
    // 1. Desenhando a malha de fundo (grid)
    ctx.beginPath();
    ctx.strokeStyle = "#e0e0e0"; // Cinza bem claro para não poluir
    ctx.lineWidth = 1;

    // Linhas verticais (cruzando o eixo X)
    for (let x = 0; x <= width; x += step) {
        ctx.moveTo(x, 0);
        ctx.lineTo(x, height);
    }

    // Linhas horizontais (cruzando o eixo Y)
    for (let y = 0; y <= height; y += step) {
        ctx.moveTo(0, y);
        ctx.lineTo(width, y);
    }
    ctx.stroke();

    ctx.beginPath();
    ctx.fillStyle = "red";
    // O raio é 5 pixels, indo de 0 a 2*PI (círculo completo)
    ctx.arc(width/2, height/2, 5, 0, Math.PI * 2);
    ctx.fill();
}

Projeção isométrica

A projeção isométrica é uma das técnicas mais utilizadas em computação gráfica para representar objetos tridimensionais em um plano bidimensional. Ela permite que as três dimensões do objeto sejam representadas de forma proporcional, mantendo a escala e a perspectiva. A seguir, apresentaremos um exemplo de código que demonstra como aplicar a projeção isométrica em um quadrado, transformando suas coordenadas para criar a ilusão de profundidade. 

/** @type {HTMLCanvasElement} */
const c = document.getElementById("myCanvas")
const ctx = c.getContext("2d")

const width = 800
const height = 800
c.width = width
c.height = height
const step = 100
// Vamos considerar que 100 pixels equivale a 1 unidade matemática
const cx = width/2
const cy = height/2

function converterCoordenadas(xUnidade, yUnidade, isIsometrica) {
    let xFinal, yFinal;

    // Aplica a transformação isométrica pura 
    const xIso = xUnidade - yUnidade;
    const yIso = (xUnidade + yUnidade) / 2;

    // Transforma para coordenadas de tela
    // INVERTENDO o eixo Y (subtraindo de 'cy')
    // para a matemática ficar natural
    xFinal = cx + (xIso * step);
    yFinal = cy - (yIso * step);

    // Retorna um objeto com as coordenadas prontas para o Canvas
    return { x: xFinal, y: yFinal };
}

function desenharQuadradoIsometrico() {
    ctx.beginPath()
    ctx.lineWidth=4
    ctx.strokeStyle = "green";
    // Pegamos as coordenadas processadas pela nossa função
    let p1 = converterCoordenadas(0, 0);
    let p2 = converterCoordenadas(1, 0);
    let p3 = converterCoordenadas(1, 1);
    let p4 = converterCoordenadas(0, 1);
    // Desenhamos usando os valores prontos, sem pensar em pixels!
    ctx.moveTo(p1.x, p1.y); // A(0, 0)
    ctx.lineTo(p2.x, p2.y); // B(1, 0)
    ctx.lineTo(p3.x, p3.y); // C(1, 1)
    ctx.lineTo(p4.x, p4.y); // D(0, 1)
    ctx.lineTo(p1.x, p1.y); // Volta para a origem fechando o polígono
    ctx.stroke();
}

function planoCartesiano() {
    ctx.beginPath()
    ctx.lineWidth = 1
    ctx.strokeStyle="black"

    for (let x=0; x<=width; x+=step) {
        ctx.moveTo(x, 0);
        ctx.lineTo(x, height)
    }

    for (let y=0; y<=height; y+=step) {
        ctx.moveTo(0, y)
        ctx.lineTo(width, y)
    }

    ctx.stroke()
}

Círculo girando

Nesta seção, vamos explorar a criação de um círculo que gira em torno de seu centro. Utilizando transformações matemáticas e a função de rotação, podemos animar o círculo para que ele gire continuamente, proporcionando uma experiência visual dinâmica. A seguir, apresentaremos o código necessário para implementar essa animação, juntamente com explicações detalhadas sobre cada parte do processo. 

/** @type { HTMLCanvasElement } */
const c = document.getElementById("myCanvas")
const ctx = c.getContext("2d")

const width = 500;
const height = 500;
c.width = width;
c.height = height;
const step = 100; // 1 unidade matemática = 100 pixels
const cx = width / 2;
const cy = height / 2;

// Variável que vai alimentar o giro do quadrado
let anguloGlobal = 30;

function converterCoordenadas(x, y, anguloGraus) {
    // 1. Converte o ângulo de graus para radianos
    const rad = anguloGraus * Math.PI / 180;

    // 2. Rotaciona as coordenadas originais XY 
    const xRot = (x * Math.cos(rad)) - (y * Math.sin(rad));
    const yRot = (x * Math.sin(rad)) + (y * Math.cos(rad));

    // 3. Aplica o "step" para dar tamanho na tela e centraliza.
    // Lembre-se: subtraímos o Y porque no Canvas ele cresce para baixo! 
    return {
        x: cx + (xRot * step),
        y: cy - (yRot * step)
    };
}

function desenharCirculoRotacionando() {
    const startX = 2
    const startY = 0
    let linha = converterCoordenadas(startX, startY, anguloGlobal)
    let circulo = converterCoordenadas(startX, startY, anguloGlobal)
    
    ctx.beginPath()
    ctx.moveTo(linha.x, linha.y)
    ctx.lineTo(width/2, height/2)
    ctx.stroke()
    
    ctx.beginPath()
    ctx.arc(circulo.x, circulo.y, 10, 0, 2 * Math.PI)
    ctx.stroke()
    ctx.closePath()
}

function desenharCirculoReferencia() {
    const raio = 200
    ctx.beginPath()
    for (let angle = 0; angle <= 360; angle=angle+30) {
        const x = width/2 + raio * Math.cos(angle * Math.PI/180)
        const y = height/2 + raio * Math.sin(angle * Math.PI/180)
        ctx.lineTo(x, y)
    }
    ctx.stroke()
}

/**
 * O LOOP DE RENDERIZAÇÃO
 */

function animar() {
    ctx.clearRect(0, 0, width, height); 
	desenharEixos();
    desenharCirculoRotacionando();
    desenharCirculoReferencia()
    anguloGlobal += 1;
    requestAnimationFrame(animar); 
}

// Dá a partida no motor!
animar();

function desenharEixos() {
    ctx.beginPath()
    ctx.moveTo(width/2, 0)
    ctx.lineTo(width/2, height)
    ctx.lineWidth = 2
    ctx.strokeStyle="red"
    ctx.stroke()
    ctx.closePath()

    ctx.beginPath()
    ctx.moveTo(0, height/2)
    ctx.lineTo(width, height/2)
    ctx.lineWidth = 2
    ctx.strokeStyle="red"
    ctx.stroke()
    ctx.closePath()
}

Construindo uma esfera

Vamos deixar de lado a rotação de objetos por enquanto, e vamos explorar a construção de uma esfera. A esfera é um objeto tridimensional que pode ser representado em um plano bidimensional através de projeções e técnicas de sombreamento. A seguir, apresentaremos um exemplo de código que demonstra como desenhar uma esfera utilizando o elemento canvas do HTML e JavaScript, criando uma representação visual que simula a aparência tridimensional da esfera. 

/** @type { HTMLCanvasElement } */
const c = document.getElementById("myCanvas")
const ctx = c.getContext("2d")

const width = 800
const height = 600
c.width = width
c.height = height
const step = 100 // Por enquanto, vamos considerar que 100 equivale a 1 unidade, ok?!
const cx = width/2
const cy = height/2

function converterCoordenadas(x, y, z, anguloGraus) {
    // 1. Converte o ângulo de graus para radianos
    const rad = anguloGraus * Math.PI / 180;

    // 2. Rotaciona as coordenadas XY (girando o objeto em torno de Z)
    const xRot = (x * Math.cos(rad)) - (y * Math.sin(rad));
    const yRot = (x * Math.sin(rad)) + (y * Math.cos(rad));

    // 3. Aplica a Projeção Isométrica 2:1 [cite: 312]
    const xIso = xRot - yRot;
    const yIso = ((xRot + yRot) / 2) + z;

    // 4. Converte para coordenadas da tela
    return {
        x: cx + (xIso * step),
        y: cy - (yIso * step)
    };
}

function desenharEsferaCompletaEstatica() {
    ctx.beginPath();
    ctx.lineWidth = 1;
    // O clássico verde fóssforo de monitores antigos
    ctx.strokeStyle = "green";

    const raioEsfera = 2; // O tamanho da nossa esfera em unidades puras

    // ========================================================
    // 1. LATITUDES (Os anéis horizontais)
    // ========================================================
    for (let phi = -90; phi <= 90; phi += 15) {
        const radPhi = phi * Math.PI / 180;
        
        // Calcula a altura Z e o raio da fatia atual
        const zAtual = raioEsfera * Math.sin(radPhi);
        const raioAtual = raioEsfera * Math.cos(radPhi);

        for (let theta = 0; theta <= 360; theta += 15) {
            const radTheta = theta * Math.PI / 180;
            
            const x = raioAtual * Math.cos(radTheta);
            const y = raioAtual * Math.sin(radTheta);
            
            const ponto = converterCoordenadas(x, y, zAtual, 0);

            if (theta === 0) {
                ctx.moveTo(ponto.x, ponto.y);
            } else {
                ctx.lineTo(ponto.x, ponto.y);
            }
        }
    }

    // ========================================================
    // 2. MERIDIANOS (As linhas verticais)
    // ========================================================
    // Laço Externo agora trava a longitude (theta)
    for (let theta = 0; theta <= 360; theta += 15) {
        const radTheta = theta * Math.PI / 180;

        // Laço Interno viaja do Polo Sul ao Polo Norte (phi)
        for (let phi = -90; phi <= 90; phi += 15) {
            const radPhi = phi * Math.PI / 180;
            
            const zAtual = raioEsfera * Math.sin(radPhi);
            const raioAtual = raioEsfera * Math.cos(radPhi);

            const x = raioAtual * Math.cos(radTheta);
            const y = raioAtual * Math.sin(radTheta);

            const ponto = converterCoordenadas(x, y, zAtual, 0);

            // Se for o Polo Sul (-90), levanta o lápis. Depois, vai riscando.
            if (phi === -90) {
                ctx.moveTo(ponto.x, ponto.y);
            } else {
                ctx.lineTo(ponto.x, ponto.y);
            }
        }
    }

    ctx.stroke();
}

// Limpa a tela e chama a renderização
ctx.clearRect(0, 0, width, height);
desenharEsferaCompletaEstatica();
